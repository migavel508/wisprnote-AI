101    #[objc::msg_send(queryPedometerDataFromDate:toDate:withHandler:)]
102    pub fn query_pedometer_data_ch_block(

105        end: &ns::Date,
106        handler: &mut blocks::ResultCh<cm::PedometerData>,
107    );
108
109    #[inline]
110    pub fn query_pedometer_data_ch(
111        &self,
112        start: &ns::Date,
113        end: &ns::Date,
114        handler: impl FnMut(Option<&cm::PedometerData>, Option<&ns::Error>)
115        + 'static
116        + std::marker::Sync,
117    ) {
118        let mut handler = blocks::ResultCh::new2(handler);
119        self.query_pedometer_data_ch_block(start, end, &mut handler)
120    }
121
122    #[cfg(feature = "async")]
123    pub async fn query_pedometer_data(
124        &self,
125        start: &ns::Date,
126        end: &ns::Date,
127    ) -> Result<arc::R<cm::PedometerData>, arc::R<ns::Error>> {
128        let (future, mut block) = blocks::result();
129        self.query_pedometer_data_ch_block(start, end, &mut block);
130        future.await
131    }
132
133    #[objc::msg_send(startPedometerUpdatesFromDate:withHandler:)]
