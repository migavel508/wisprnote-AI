302    fn hid_header() -> String {
303        let output = Command::new("xcrun")
304            .args(["--sdk", "macosx", "--show-sdk-path"])
305            .output()
306            .expect("xcrun should be available to locate the macOS SDK");
307        assert!(
308            output.status.success(),
309            "xcrun failed: {}",
310            String::from_utf8_lossy(&output.stderr)
311        );
312        let sdk = String::from_utf8(output.stdout).expect("xcrun output should be utf-8");
313        let path = format!(
314            "{}/System/Library/Frameworks/IOKit.framework/Headers/hid/IOHIDUsageTables.h",
315            sdk.trim()
316        );
317        let bytes = fs::read(path).expect("IOHIDUsageTables.h should exist in the SDK");
318        String::from_utf8_lossy(&bytes).into_owned()
319    }

321    fn hid_usage(header: &str, symbol: &str) -> KeyCode {
322        let line = header
323            .lines()
324            .find(|line| {
325                line.split('=')
326                    .next()
327                    .map(str::trim)
328                    .is_some_and(|name| name == symbol)
329            })
330            .unwrap_or_else(|| panic!("missing {symbol} in IOHIDUsageTables.h"));
331        let value = line
332            .split('=')
333            .nth(1)
334            .unwrap_or_else(|| panic!("missing value for {symbol}"))
335            .split(',')
336            .next()
337            .unwrap()
338            .trim();
339
340        if value.starts_with("0x") {
341            KeyCode::from_str_radix(value.trim_start_matches("0x"), 16)
342                .expect("hex value should parse")
343        } else {
344            value.parse::<KeyCode>().expect("value should parse")
345        }
346    }
