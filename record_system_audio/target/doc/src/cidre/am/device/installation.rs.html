5impl<'a> Session<'a> {
6    /// Lookup the set of installed applications on the device.
7    ///
8    /// The info argument will point to a cf::Dictionary. Each key in the dictionary is an application
9    /// identifier and the value is the Info.plist of the installed application.
10    ///
11    /// The client can specify options to filter the results.
12    //// If the value of kLookupApplicationTypeKey is:
13    ////     kApplicationTypeAny: All application types are returned.
14    ////     kApplicationTypeSystem: Only applications installed with the OS are returned.
15    ////     kApplicationTypeUser: Only third party applications are returned.
16    ////
17    /// If the value of kLookupAttributeKey is:
18    ////    kAttributeHasSettingsBundle: Only applications with the "HasSettingsBundle" attribute are returned.
19    ////     kAttributeHasSettingsBundleIcon: Only applications with the "HasSettingsBundleIcon" attribute are returned.
20    ////
21    ////     kLookupBundleIDsKey
22    ////         CFString or array of strings that indicate the bundle IDs to lookup
23    ////     kLookupReturnAttributesKey
24    ////         CFString or array of strings indicating which of the keys from the Info.plist should be returned
25    ////
26    /// IMPORTANT NOTE:
27    /// You should specify kLookupReturnAttributesKey in the options dictionary with an array value containing those
28    /// keys you would like returned. If you do not specify this attribute, MobileDevice will log every time you make
29    /// an offending call and performance will be poor. When specifying this key, you MUST include at least
30    /// kCFBundleIdentifierKey in the array of keys to be returned or the call will not work properly.
31    pub fn lookup_apps(&self, options: &cf::Dictionary) -> Result<arc::R<cf::Dictionary>, Error> {
32        unsafe {
33            let mut info = None;
34            AMDeviceLookupApplications(self, options, &mut info).to_result(info)
35        }
36    }

53impl<'a> AppsLookupBuilder<'a> {

83
84    pub fn lookup(&self) -> Result<arc::R<cf::Dictionary>, Error> {
85        let mut options = cf::DictionaryMut::with_capacity(3);
86        if let Some(ref app_type) = self.application_type {
87            options.insert(cf::str!(c"ApplicationType"), app_type);
88        }
89
90        if let Some(ref attribute) = self.attribute {
91            options.insert(cf::str!(c"Attribute"), attribute)
92        }
93
94        self.session.lookup_apps(&options)
95    }
96}
