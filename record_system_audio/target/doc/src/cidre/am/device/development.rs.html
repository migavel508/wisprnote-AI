7/// Image Mount Callback
8/// Ownership of the status dictionary *status* passes to the callback function. The dict MUST BE
9/// explicitly released or else it will leak.
10pub type MountCallback<T> = extern "C" fn(status: arc::R<cf::Dictionary>, context: *mut T);

13    fn AMDeviceMountImage(

18        context: *mut c_void,
19    ) -> Error;

22}

24pub fn xcode_dev_path() -> PathBuf {
25    use std::process::Command;
26    let command = Command::new("xcode-select")
27        .arg("-print-path")
28        .output()
29        .expect("xcode-select prints path");
30    String::from_utf8(command.stdout)
31        .expect("valid utf-8 output from xcode-select command")
32        .trim()
33        .into()
34}

44    pub unsafe fn mound_disk_with_callback<T>(
45        &self,
46        image: &cf::String,
47        options: &cf::Dictionary,
48        callback: *const MountCallback<T>,

54    pub fn mount_developer_image(&self) -> Result<(), Error> {
55        let ds_path = self.device_support_path();
56        if ds_path.is_none() {
57            return Err(Error::NOT_FOUND);
58        }
59        let ds_path = ds_path.unwrap();
60        let image_path = ds_path.join("DeveloperDiskImage.dmg");
61        let sig_image_path = ds_path.join("DeveloperDiskImage.dmg.signature");
62        let sig = std::fs::read(sig_image_path).expect("sig file read");
63        let sig: arc::R<cf::Data> = arc::R::from(&sig[..]);
64
65        let image_sig_key = cf::String::from_str("ImageSignature");
66        let options = cf::Dictionary::with_keys_values(
67            &[&image_type::key(), &image_sig_key],
68            &[&image_type::developer(), &sig],
69        )
70        .expect("options for mount created");
71
72        let path = image_path.to_str().unwrap();
73        let cf_image_path = &cf::String::from_str(path);
74        self.mound_disk(cf_image_path, &options)
75    }
76
77    pub fn mound_disk(
78        &self,
79        image_path: &cf::String,
80        options: &cf::Dictionary,
81    ) -> Result<(), Error> {
82        extern "C" fn mount_cb(status: arc::R<cf::Dictionary>, _context: *mut c_void) {
83            status.show();
84        }
85        unsafe {
86            self.mound_disk_with_callback::<c_void>(
87                image_path,
88                options,
89                mount_cb as _,
90                std::ptr::null_mut(),
91            )
92            .result()
93        }
94    }
95
96    /// Copy a file from the relay service.
97    /// Options must contain:
98    ///
99    /// kAMDRelayTypeKey: one of
100    ///     kAMDRelayTypeFileDescriptor
101    ///     kAMDRelayTypeData
102    ///
103    //// kAMDRelayLocationKey:
104    ////     for kAMDRelayTypeFileDescriptor, this must be a CFNumber containing a file descriptor
105    ////     on Windows, this must be a *file descriptor* that can be passed
106    ////     to the 'write' call; this is not the same as a Windows HANDLE.
107    ////
108    ////     for kAMDRelayTypeData, this must be a CFMutableData; the relayed stream will be appended
109    ////     to that CFMutableData
110    pub fn relay_file(&self, source: &cf::String, options: &cf::Dictionary) -> Error {
111        unsafe { AMDeviceRelayFile(self, source, options) }
112    }

115pub fn platform_support_path(platform: &str, os_version: &str) -> Option<PathBuf> {
116    let prefix = xcode_dev_path()
117        .join("Platforms")
118        .join(platform)
119        .join("DeviceSupport");
120    let version = os_version
121        .splitn(3, '.')
122        .take(2)
123        .map(|s| u32::from_str_radix(s, 10).unwrap_or_default())
124        .collect::<Vec<_>>();
125    let major = version[0].to_string();
126    let mut minor = version[1];
127
128    let version = format!("{major}.{minor}");
129
130    let mut matched_majors = Vec::new();
131
132    for directory in std::fs::read_dir(&prefix).expect("folder exists") {
133        let directory = directory.expect("folder exists");
134        let name = directory.file_name().into_string().expect("valid string");
135        if name.starts_with(&version) {
136            return Some(prefix.join(name));
137        }
138        if name.starts_with(&major) {
139            matched_majors.push(name);
140        }
141    }
142
143    while minor > 0 {
144        let version = format!("{major}.{minor}");
145        if matched_majors.contains(&version) {
146            return Some(prefix.join(version));
147        }
148        minor -= 1;
149    }
150
151    None
152}
