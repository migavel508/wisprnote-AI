1use crate::cm::io;
2
3/// CMIOSystemObject Properties
4/// CMIOObjectPropertySelector values that apply to the CMIOSystemObject.
5///
6/// The CMIOSystemObject has one scope, kCMIOObjectPropertyScopeGlobal, and only a main element.
7impl io::PropSelector {
8    /// A u32 where 1 means that the current process contains the main instance of the DAL.
9    /// The main instance of the DAL is the only instance in which plug-ins should save/restore their devices' settings.
10    #[doc(alias = "kCMIOHardwarePropertyProcessIsMain")]
11    pub const HW_PROCESS_IS_MAIN: Self = Self(u32::from_be_bytes(*b"main"));
12
13    /// A u32 whose value will be non-zero if the DAL is either in the midst of initializing or in the midst of exiting the process.
14    #[doc(alias = "kCMIOHardwarePropertyIsInitingOrExiting")]
15    pub const HW_IS_INITING_OR_EXITING: Self = Self(u32::from_be_bytes(*b"inot"));
16
17    /// An array of the CMIODeviceIDs that represent all the devices currently available to the system.
18    #[doc(alias = "kCMIOHardwarePropertyDevices")]
19    pub const HW_DEVICES: Self = Self(u32::from_be_bytes(*b"dev#"));
20
21    /// The CMIODeviceID of the default input CMIODevice.
22    #[doc(alias = "kCMIOHardwarePropertyDefaultInputDevice")]
23    pub const HW_DEFAULT_INPUT_DEVICE: Self = Self(u32::from_be_bytes(*b"dIn "));
24
25    /// The CMIODeviceID of the default output CMIODevice.
26    #[doc(alias = "kCMIOHardwarePropertyDefaultOutputDevice")]
27    pub const HW_DEFAULT_OUTPUT_DEVICE: Self = Self(u32::from_be_bytes(*b"dOut"));
28
29    /// Using a AudioValueTranslation structure, this property translates the input CFStringRef containing a UID into the CMIODeviceID
30    /// that refers to the CMIODevice with that UID. This property will return kCMIODeviceUnknown if the given UID does not match
31    /// any currently available CMIODevice.
32    #[doc(alias = "kCMIOHardwarePropertyDeviceForUID")]
33    pub const HW_DEVICE_FOR_UID: Self = Self(u32::from_be_bytes(*b"duid"));
34
35    /// A u32 where 1 means that the process will allow the CPU to idle sleep even if there is IO in progress.
36    /// A 0 means that the CPU will not be allowed to idle
37    #[doc(alias = "kCMIOHardwarePropertySleepingIsAllowed")]
38    pub const HW_SLEEPING_IS_ALLOWED: Self = Self(u32::from_be_bytes(*b"slep"));
39
40    /// A u32 where 1 means that this process wants the DAL to unload itself after a period of inactivity where there are
41    /// no streams active and no listeners registered with any CMIOObject.
42    #[doc(alias = "kCMIOHardwarePropertyUnloadingIsAllowed")]
43    pub const HW_UNLOADING_IS_ALLOWED: Self = Self(u32::from_be_bytes(*b"unld"));
44
45    /// Using a AudioValueTranslation structure, this property translates the input CFString containing a bundle ID into the CMIOObjectID of the CMIOPlugIn that
46    /// corresponds to it. This property will return kCMIOObjectUnkown if the given bundle ID doesn't match any CMIOPlugIns.
47    #[doc(alias = "kCMIOHardwarePropertyPlugInForBundleID")]
48    pub const HW_PLUG_IN_FOR_BUNDLE_ID: Self = Self(u32::from_be_bytes(*b"pibi"));
49
50    /// A u32 where a value other than 0 indicates that the login session of the user of the process is either an active console session
51    /// or a headless session.
52    #[doc(alias = "kCMIOHardwarePropertyUserSessionIsActiveOrHeadless")]
53    pub const HW_USER_SESSION_IS_ACTIVE_OR_HEADLESS: Self = Self(u32::from_be_bytes(*b"user"));
54
55    /// A u32 where a value of 0 indicates the hardware is not suspended due to a system action, and a value of 1 means that it is.
56    /// For example, if a fast user switch occurs, the system will suspend all devices. While suspended, no operartions can be performed on any devices.
57    /// This property is never settable.
58    #[doc(alias = "kCMIOHardwarePropertySuspendedBySystem")]
59    pub const HW_SUSPENDED_BY_SYSTEM: Self = Self(u32::from_be_bytes(*b"sbys"));
60
61    /// A u32 where 1 means that screen capture devices will be presented to the process.
62    /// A 0 means screen capture devices will be ignored. By default, this property is 1.
63    #[doc(alias = "kCMIOHardwarePropertyAllowScreenCaptureDevices")]
64    pub const HW_ALLOW_SCREEN_CAPTURE_DEVICES: Self = Self(u32::from_be_bytes(*b"yes "));
65
66    /// A u32 where 1 means that wireless screen capture devices will be presented to the process.
67    /// A 0 means wireless screen capture devices will be ignored. By default, this property is 0.
68    #[doc(alias = "kCMIOHardwarePropertyAllowWirelessScreenCaptureDevices")]
69    pub const HW_ALLOW_WIRELESS_SCREEN_CAPTURE_DEVICES: Self = Self(u32::from_be_bytes(*b"wscd"));
70}
