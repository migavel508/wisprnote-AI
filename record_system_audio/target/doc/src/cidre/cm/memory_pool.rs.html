10/// Memory pool for optimizing repeated large block allocation.
11///
12/// cm::MemoryPool is a memory allocation service that holds onto a pool of
13/// recently deallocated memory so as to speed up subsequent allocations of the same size.  
14/// It's intended for cases where large memory blocks need to be repeatedly allocated --
15/// for example, the compressed data output by a video encoder.
16///
17/// All of its allocations are on the granularity of page sizes; it does not suballocate
18/// memory within pages, so it is a poor choice for allocating tiny blocks.
19/// For example, it's appropriate to use as the blockAllocator argument to
20/// cm::BlockBufferCreateWithMemoryBlock, but not the structureAllocator argument --
21/// use kCFAllocatorDefault instead.
22/// When you no longer need to allocate memory from the pool, call cm::MemoryPoolInvalidate
23/// and CFRelease. Calling CMMemoryPoolInvalidate tells the pool to stop holding onto
24/// memory for reuse.  Note that the pool's cf::Allocator can outlive the pool, owing
25/// to the way that CoreFoundation is designed: cf::Allocators are themselves CF objects,
26/// and every object allocated with a cf::Allocator implicitly retains the cf::Allocator
27/// until it is finalized.  After the cm::MemoryPool is invalidated or finalized,
28/// its cf::Allocator allocates and deallocates with no pooling behavior.
29///
30/// cm::MemoryPool deallocates memory if it has not been recycled in 0.5 second,
31/// so that short-term peak usage does not cause persistent bloat.
32/// (This period may be overridden by specifying kCMMemoryPoolOption_AgeOutPeriod.)
33/// Such "aging out" is done during the pool's cf::Allocator::allocate and
34/// cf::Allocator::deallocate methods.
35impl MemPool {
36    ///```
37    /// use cidre::cm;
38    /// let mut pool = cm::MemPool::new();
39    /// let allocator = pool.pool_allocator();
40    /// pool.flush();
41    ///````
42    #[doc(alias = "CMMemoryPoolCreate")]
43    #[inline]
44    pub fn new() -> arc::R<Self> {
45        Self::with_opts(None)
46    }
47
48    #[doc(alias = "CMMemoryPoolCreate")]
49    #[inline]
50    pub fn with_opts(options: Option<&cf::Dictionary>) -> arc::R<MemPool> {
51        unsafe { CMMemoryPoolCreate(options) }
52    }
53
54    #[inline]
55    pub fn with_age(duration: std::time::Duration) -> arc::R<Self> {
56        let opts = cf::Dictionary::with_keys_values(
57            &[keys::age_out_period()],
58            &[&cf::Number::from_f64(duration.as_secs_f64())],
59        );
60
61        Self::with_opts(opts.as_deref())
62    }
63
64    #[doc(alias = "CMMemoryPoolGetAllocator")]
65    #[inline]
66    pub fn pool_allocator(&self) -> &cf::Allocator {
67        unsafe { CMMemoryPoolGetAllocator(self) }
68    }
69
70    /// Deallocates all memory the pool was holding for recycling.
71    #[doc(alias = "CMMemoryPoolFlush")]
72    #[inline]
73    pub fn flush(&mut self) {
74        unsafe { CMMemoryPoolFlush(self) }
75    }
76
77    /// Stops the pool from recycling.
78    #[doc(alias = "CMMemoryPoolInvalidate")]
79    #[inline]
80    pub fn invalidate(&mut self) {
81        unsafe { CMMemoryPoolInvalidate(self) }
82    }
83}
