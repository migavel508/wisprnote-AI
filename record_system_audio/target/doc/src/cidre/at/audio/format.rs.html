51pub enum BalanceFadeType {

53    /// This can reduce overall loudness when the balance or fade is not in the center.
54    MaxUnityGain = 0,

56    /// The overall loudness remains constant, but gain can exceed 1.0.

252impl PropId {
253    pub unsafe fn info(
254        &self,
255        in_specifier_size: u32,
256        in_specifier: *const c_void,
257    ) -> os::Result<u32> {
258        let mut out_property_data_size = 0;
259        unsafe {
260            AudioFormatGetPropertyInfo(
261                *self,
262                in_specifier_size,
263                in_specifier,
264                &mut out_property_data_size,
265            )
266            .result()
267        }?;
268        Ok(out_property_data_size)
269    }
270
271    pub unsafe fn value(
272        &self,
273        in_specifier_size: u32,
274        in_specifier: *const c_void,
275        io_property_data_size: *mut u32,
276        out_property_data: *mut c_void,
277    ) -> os::Result {
278        unsafe {
279            AudioFormatGetProperty(
280                *self,
281                in_specifier_size,
282                in_specifier,
283                io_property_data_size,
284                out_property_data,
285            )
286            .result()
287        }
288    }
289
290    pub unsafe fn fill<T: Sized>(&self, val: &mut T) -> os::Result {
291        let mut size = size_of::<T>() as u32;
292        unsafe { self.value(0, std::ptr::null(), &mut size, val as *mut _ as *mut _) }
293    }
294
295    pub unsafe fn fill_with<S: Sized, T: Sized>(&self, val: &mut T, specifier: &S) -> os::Result {
296        let mut size = size_of::<T>() as u32;
297        let spec_size = size_of::<S>() as u32;
298        unsafe {
299            self.value(
300                spec_size,
301                specifier as *const _ as _,
302                &mut size,
303                val as *mut _ as *mut _,
304            )
305        }
306    }
307
308    pub unsafe fn get_vec<T: Sized>(&self) -> os::Result<Vec<T>> {
309        let mut size = unsafe { self.info(0, std::ptr::null()) }?;
310        let len = size as usize / size_of::<T>();
311        let mut result = Vec::with_capacity(len);
312        unsafe { self.value(0, std::ptr::null(), &mut size, result.as_mut_ptr() as _) }?;
313        unsafe { result.set_len(len) };
314        Ok(result)
315    }
316
317    pub unsafe fn get_vec_with<S: Sized, T: Sized>(&self, specifier: &S) -> os::Result<Vec<T>> {
318        let spec_size = size_of::<S>() as u32;
319        let mut size = unsafe { self.info(spec_size, specifier as *const _ as _) }?;
320        let len = size as usize / size_of::<T>();
321        let mut result = Vec::with_capacity(len);
322        unsafe {
323            self.value(
324                spec_size,
325                specifier as *const _ as _,
326                &mut size,
327                result.as_mut_ptr() as _,
328            )
329        }?;
330        unsafe { result.set_len(len) };
331        Ok(result)
332    }
