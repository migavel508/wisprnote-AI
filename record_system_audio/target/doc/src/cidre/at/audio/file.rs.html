353        let (size, _writable) = self.property_info(prop_id)?;
354        let mut sz: u32 = size as _;
355        let mut vec = Vec::with_capacity(size / std::mem::size_of::<T>());

357        unsafe { vec.set_len(sz as usize / std::mem::size_of::<T>()) };
358        Ok(vec)

706#[cfg(test)]
707mod tests {
708    use crate::{at::audio, cf};
709
710    #[test]
711    fn basics() {
712        let path = cf::Url::from_str("file:///tmp/m4a.m4a").unwrap();
713
714        let asbd = audio::StreamBasicDesc {
715            format: audio::Format::MPEG4_AAC,
716            format_flags: audio::FormatFlags::ALL_CLEAR,
717            frames_per_packet: 1024,
718            sample_rate: 48_000.0,
719            channels_per_frame: 2,
720            ..Default::default()
721        };
722
723        let mut file = audio::FileId::create(
724            &path,
725            audio::FileTypeId::M4A,
726            &asbd,
727            audio::FileFlags::ERASE_FILE,
728        )
729        .unwrap();
730
731        let (size, writable) = file
732            .property_info(audio::FilePropId::DEFER_SIZE_UPDATES)
733            .unwrap();
734
735        assert_eq!(size, std::mem::size_of::<u32>());
736        assert_eq!(writable, true);
737
738        let defer_size_updates = file.defer_size_updates().unwrap();
739
740        assert!(defer_size_updates);
741
742        file.set_defer_size_updates(false).unwrap();
743
744        let defer_size_updates = file.defer_size_updates().unwrap();
745
746        assert!(!defer_size_updates);
747
748        assert_eq!(file.file_format().unwrap(), audio::FileTypeId::M4A);
749
750        let (size, writable) = file.property_info(audio::FilePropId::FILE_FORMAT).unwrap();
751
752        assert_eq!(size, std::mem::size_of::<audio::FileTypeId>());
753        assert_eq!(writable, false);
754
755        let (size, writable) = file.property_info(audio::FilePropId::DATA_FORMAT).unwrap();
756
757        assert_eq!(size as usize, std::mem::size_of::<audio::StreamBasicDesc>());
758        assert_eq!(writable, true);
759
760        let file_asbd = file.data_format().unwrap();
761
762        assert_eq!(file_asbd, asbd);
763
764        file.set_data_format(&asbd).unwrap();
765
766        let (size, writable) = file.property_info(audio::FilePropId::IS_OPTIMIZED).unwrap();
767
768        assert_eq!(size, std::mem::size_of::<u32>());
769        assert_eq!(writable, false);
770
771        assert_eq!(file.optimized().unwrap(), false);
772
773        assert!(
774            file.property_info(audio::FilePropId::DATA_FORMAT_NAME)
775                .is_err()
776        );
777
778        let (size, writable) = file
779            .property_info(audio::FilePropId::RESERVE_DURATION)
780            .unwrap();
781
782        assert_eq!(size, std::mem::size_of::<f64>());
783        assert_eq!(writable, true);
784
785        file.set_reserve_duration(1000f64).unwrap();
786
787        let (size, writable) = file
788            .property_info(audio::FilePropId::ESTIMATED_DURATION)
789            .unwrap();
790
791        assert_eq!(size, std::mem::size_of::<f64>());
792        assert_eq!(writable, false);
793
794        assert_eq!(0f64, file.estimated_duration().unwrap());
795
796        let (size, writable) = file
797            .property_info(audio::FilePropId::INFO_DICTIONARY)
798            .unwrap();
799
800        assert_eq!(size, std::mem::size_of::<usize>());
801        assert_eq!(writable, false);
802
803        let info = file.info_dictionary().unwrap();
804
805        assert!(!info.is_empty());
806        info.show();
807
808        _ = file.close();
809        std::mem::forget(file);
810    }
