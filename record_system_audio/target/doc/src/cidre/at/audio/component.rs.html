10/// AudioComponentFlags
11impl Flags {
12    /// When this bit in AudioComponentDescription's componentFlags is set, AudioComponentFindNext
13    /// will only return this component when performing a specific, non-wildcard search for the
14    /// component, i.e. with non-zero values of componentType, componentSubType, and
15    /// componentManufacturer. This can be useful when privately registering a component.
16    pub const UNSEARCHABLE: Self = Self(1);
17
18    /// An AudioComponent sets this bit in its componentFlags to indicate to the system that the
19    /// AudioComponent is safe to open in a sandboxed process.
20    pub const SANDBOX_SAFE: Self = Self(2);
21
22    /// The system sets this flag automatically when registering components which implement a version 3
23    /// Audio Unit.
24    pub const IS_V3_AUDIO_UNIT: Self = Self(4);
25
26    /// The system sets this flag automatically when registering components which require asynchronous
27    /// instantiation via AudioComponentInstantiate (v3 audio units with views).
28    pub const REQUIRES_ASYNC_INSTANTIATION: Self = Self(8);
29
30    /// The system sets this flag automatically when registering components which can be loaded into
31    /// the current process. This is always true for V2 audio units; it depends on the packaging
32    /// in the case of a V3 audio unit.
33    pub const CAN_LOAD_IN_PROCESS: Self = Self(0x10);
34}

36define_opts!(pub InstantiationOpts(u32));
37
38/// AudioComponentInstantiationOptions
39///
40/// Most component instances are loaded into the calling process.
41///
42/// A version 3 audio unit, however, can be loaded into a separate extension service process,
43/// and this is the default behavior for these components. To be able to load one in-process
44/// requires that the developer package the audio unit in a bundle separate from the application
45/// extension, since an extension's main binary cannot be dynamically loaded into another
46/// process.
47///
48/// A macOS host may request in-process loading of such audio units using
49/// LOAD_IN_PROCESS.
50///
51/// Flag::IS_V3_AUDIO_UNIT specifies whether an audio unit is implemented using API
52/// version 3.
53///
54/// These options are just requests to the implementation. It may fail and fall back to the
55/// default.
56impl InstantiationOpts {
57    /// Attempt to load the component into a separate extension process.
58    pub const LOAD_OUT_OF_PROCESS: Self = Self(1);
59
60    /// Attempt to load the component into the current process. Only available on macOS.
61    pub const LOAD_IN_PROCESS: Self = Self(2);
62    pub const LOADED_REMOTELY: Self = Self(1 << 31);
63}

65/// A structure describing the unique and identifying IDs of an audio component
66#[doc(alias = "AudioComponentDescription")]
67#[derive(Default)]
68#[repr(C)]
69pub struct Desc {
70    /// A 4-char code identifying the generic type of an audio component.
71    /// `aenc` for example
72    pub type_: os::Type,
73
74    /// A 4-char code identifying the a specific individual component. type/
75    /// subtype/manufacturer triples must be globally unique.
76    /// `aac ` for example
77    pub sub_type: os::Type,
78
79    /// Vendor identification.
80    pub manufacturer: os::Type,
81
82    /// Must be set to zero unless a known specific value is requested.
83    pub flags: u32,
84
85    /// Must be set to zero unless a known specific value is requested.
86    pub flags_mask: u32,
87}
88
89impl std::fmt::Debug for Desc {
90    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
91        let mut type_ = self.type_.to_be_bytes();
92        let type_ = crate::four_cc_to_str(&mut type_);
93
94        let mut sub_type = self.sub_type.to_be_bytes();
95        let sub_type = crate::four_cc_to_str(&mut sub_type);
96
97        let mut manufacturer = self.manufacturer.to_be_bytes();
98        let manufacturer = crate::four_cc_to_str(&mut manufacturer);
99        f.debug_struct("AudioComponentDescription")
100            .field("type_", &type_)
101            .field("sub_type", &sub_type)
102            .field("manufacturer", &manufacturer)
103            .field("flags", &self.flags)
104            .field("flags_mask", &self.flags_mask)
105            .finish()
106    }
107}

171impl Component {
172    #[inline]
173    pub fn name(&self) -> os::Result<arc::R<cf::String>> {
174        unsafe { os::result_unchecked(|val| AudioComponentCopyName(self, val)) }
175    }
176
177    #[inline]
178    pub fn desc(&self) -> os::Result<Desc> {
179        let mut desc = Desc::default();
180        unsafe { AudioComponentGetDescription(self, &mut desc).result()? };
181        Ok(desc)
182    }
183
184    #[inline]
185    pub fn version(&self) -> os::Result<u32> {
186        let mut version = 0;
187        unsafe { AudioComponentGetVersion(self, &mut version).result()? };
188        Ok(version)
189    }
190
191    #[doc(alias = "AudioComponentInstanceNew")]
192    pub fn open(&self) -> os::Result<InstanceRef> {
193        let mut instance = None;
194        unsafe {
195            AudioComponentInstanceNew(self, &mut instance).result()?;
196            Ok(InstanceRef(instance.unwrap_unchecked()))
197        }
198    }
199}
