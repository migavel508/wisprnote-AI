103#[cfg(test)]
104mod tests {
105    use crate::{cg, ci, ns};
106
107    #[test]
108    fn basics() {
109        let vec = ci::Vec::with_xy(1.0, 2.0);
110        let point = vec.cg_point();
111        assert_eq!(vec.len(), 2);
112
113        assert_eq!(point, cg::Point { x: 1.0, y: 2.0 });
114
115        let str = vec.ns_string();
116        assert_eq!(ns::str!(c"[1 2]"), &str);
117
118        let rect = vec.cg_rect();
119        assert_eq!(
120            rect,
121            cg::Rect {
122                origin: point,
123                size: cg::Size::zero()
124            }
125        );
126
127        let _t = vec.cg_affine_transform();
128
129        assert_eq!(0.0f64, vec.get(1000));
130
131        let vec = ci::Vec::with_ns_string(ns::str!(c"hello"));
132        assert_eq!(vec.x(), 0.0f64);
133    }
