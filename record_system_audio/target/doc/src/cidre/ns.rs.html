309pub fn log_string(str: &crate::ns::String) {
310    unsafe {
311        cidre_log(str);
312    }
313}
314
315#[macro_export]
316macro_rules! log {
317    ($($arg:tt)*) => {{
318        let rstr = format!("{}",format_args!($($arg)*));
319        let ns_str = $crate::ns::String::with_str_no_copy(&rstr);
320        $crate::ns::log_string(ns_str.as_ref());
321    }};
322}
