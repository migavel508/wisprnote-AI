42#[cfg(test)]
43mod tests {
44    use crate::{nl, ns};
45    #[test]
46    fn basics() {
47        assert!(nl::LangRecognizer::dominant_lang_for_string(ns::str!(c"")).is_none());
48
49        let lang = nl::LangRecognizer::dominant_lang_for_string(ns::str!(c"un verre de cidre"))
50            .expect("failed to recognize language");
51        let fr_str = nl::Lang::with_string(ns::str!(c"fr"));
52        assert_eq!(fr_str, &lang);
53
54        let mut recognizer = nl::LangRecognizer::new();
55
56        assert!(recognizer.dominant_lang().is_none());
57
58        assert!(recognizer.lang_hypotheses(10).is_empty());
59
60        recognizer.process_string(ns::str!(c"un verre de cidre"));
61        assert_eq!(fr_str, &recognizer.dominant_lang().unwrap());
62        assert_eq!(10, recognizer.lang_hypotheses(10).len());
63
64        assert!(recognizer.lang_hints().is_empty());
65    }
