3#[doc(alias = "MTLCompareFunction")]
4#[derive(Debug, Copy, Clone, Eq, PartialEq)]
5#[repr(usize)]
6pub enum CompareFn {
7    /// A new value never passes the comparison test.
8    Never = 0,
9    /// A new value passes the comparison test if it is less than the existing value.
10    Less = 1,
11    /// A new value passes the comparison test if it is equal to the existing value.
12    Equal = 2,
13    /// A new value passes the comparison test if it is less than or equal to the existing value.
14    LessEqual = 3,
15    /// A new value passes the comparison test if it is greater than the existing value.
16    Greater = 4,
17    /// A new value passes the comparison test if it is not equal to the existing value.
18    NotEqual = 5,
19    /// A new value passes the comparison test if it is greater than or equal to the existing value.
20    GreaterEqual = 6,
21    /// A new value always passes the comparison test.
22    Always = 7,
23}

25#[doc(alias = "MTLStencilOperation")]
26#[derive(Debug, Copy, Clone, Eq, PartialEq)]
27#[repr(usize)]
28pub enum StencilOp {
29    /// Keep the current stencil value.
30    Keep = 0,
31    /// Set the stencil value to zero.
32    Zero = 1,
33    /// Replace the stencil value with the stencil reference value, which is set by
34    /// the set_stencil_reference_value method of mtl::RenderCmdEncoder.
35    Replace = 2,
36    /// If the current stencil value is not the maximum representable value, increase the stencil
37    /// value by one. Otherwise, if the current stencil value is the maximum representable value,
38    /// do not change the stencil value.
39    IncrementClamp = 3,
40    /// If the current stencil value is not zero, decrease the stencil value by one. Otherwise,
41    /// if the current stencil value is zero, do not change the stencil value.
42    DecrementClamp = 4,
43    /// Perform a logical bitwise invert operation on the current stencil value.
44    Invert = 5,
45    /// If the current stencil value is not the maximum representable value, increase
46    /// the stencil value by one. Otherwise, if the current stencil value is the maximum
47    /// representable value, set the stencil value to zero.
48    IncrementWrap = 6,
49    /// If the current stencil value is not zero, decrease the stencil value by one.
50    /// Otherwise, if the current stencil value is zero, set the stencil value to the maximum
51    /// representable value.
52    DecrementWrap = 7,
53}
