33    /// Set a global buffer for all compute kernels at the given bind point index.
34    ///
35    /// For buffers in the device address space, align the offset to the data type
36    /// consumed by the compute function (which is always less than or equal to 16 bytes).
37    ///
38    /// For buffers in the constant address space, align the offset to 256 bytes in macOS.
39    /// In iOS, align the offset to the maximum of either the data type consumed by
40    /// the compute function, or 4 bytes. A 16-byte alignment is safe in iOS if you don't
41    /// need to consider the data type.
42    ///
43    /// # Arguments
44    ///
45    /// * `buffer` - The mtl::Buf object to set in the argument table.`
46    /// * `offset` - Where the data begins, in bytes, from the start of the buffer.
47    /// * `index` - An index in the buffer argument table.
48    #[objc::msg_send(setBuffer:offset:atIndex:)]
49    pub fn set_buf_at(&mut self, buffer: Option<&mtl::Buf>, offset: usize, index: usize);
50
51    /// Set the offset within the current global buffer for all compute
