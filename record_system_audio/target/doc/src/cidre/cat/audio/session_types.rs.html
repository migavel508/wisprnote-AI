4#[doc(alias = "AVAudioSessionErrorCode")]
5#[derive(Copy, Clone, Debug, PartialEq, Eq)]
6#[repr(transparent)]
7pub struct ErrorCode(pub crate::cat::AudioInteger);
8
9impl ErrorCode {
10    /// Operation succeeded.
11    pub const NONE: Self = Self(0);
12
13    /// The app attempted to use the audio session during or after a Media Services failure.  App
14    /// should wait for a AVAudioSessionMediaServicesWereResetNotification and then rebuild all
15    /// its state.
16    pub const MEDIA_SERVICES_FAILED: Self = Self(u32::from_be_bytes(*b"msrv") as _);
17
18    /// The app attempted to set its audio session inactive or change its AVAudioSessionIOType,
19    /// but it is still actively playing and/or recording.
20    pub const IS_BUSY: Self = Self(u32::from_be_bytes(*b"!act") as _);
21
22    /// The app tried to perform an operation on a session but its category does not support it.
23    /// For instance, if the app calls setPreferredInputNumberOfChannels: while in a playback-only
24    /// category.
25    pub const INCOMPATIBLE_CATEGORY: Self = Self(u32::from_be_bytes(*b"!cat") as _);
26
27    /// The app's audio session is non-mixable and trying to go active while in the background.
28    /// This is allowed only when the app is the NowPlaying app.
29    pub const CANNOT_INTERRUPT_OTHERS: Self = Self(u32::from_be_bytes(*b"!int") as _);
30
31    /// The app does not have the required entitlements to perform an operation.
32    pub const MISSING_ENTITLEMENT: Self = Self(u32::from_be_bytes(*b"ent?") as _);
33
34    /// The app tried to do something with the audio session that is not allowed while Siri is
35    /// recording.
36    pub const SIRI_IS_RECORDING: Self = Self(u32::from_be_bytes(*b"siri") as _);
37
38    /// The app is not allowed to start recording and/or playing, usually because of a lack of audio
39    /// key in its Info.plist.  This could also happen if the app has this key but uses a category
40    /// that can't record and/or play in the background (AVAudioSessionCategoryAmbient,
41    /// AVAudioSessionCategorySoloAmbient, etc.).
42    pub const CANNOT_START_PLAYING: Self = Self(u32::from_be_bytes(*b"!pla") as _);
43
44    /// The app is not allowed to start recording, usually because it is starting a mixable
45    /// recording from the background and is not an Inter-App Audio app.
46    pub const CANNOT_START_RECORDING: Self = Self(u32::from_be_bytes(*b"!rec") as _);
47
48    /// An illegal value was used for a property.
49    pub const BAD_PARAM: Self = Self(-50);
50
51    /// The app was not allowed to set the audio category because another app (Phone, etc.) is
52    /// controlling it.
53    pub const INSUFFICIENT_PRIORITY: Self = Self(u32::from_be_bytes(*b"!pri") as _);
54
55    /// The operation failed because the device does not have sufficient hardware resources to
56    /// complete the action. For example, the operation requires audio input hardware, but the
57    /// device has no audio input available.
58    pub const RESOURCE_NOT_AVAILABLE: Self = Self(u32::from_be_bytes(*b"!res") as _);
59
60    /// An unspecified error has occurred.
61    pub const UNSPECIFIED: Self = Self(u32::from_be_bytes(*b"what") as _);
62
63    /// The operation failed because the associated session has been destroyed.
64    pub const EXPIRED_SESSION: Self = Self(u32::from_be_bytes(*b"!ses") as _);
65
66    /// The operation failed because the session is not active.
67    pub const SESSION_NOT_ACTIVE: Self = Self(u32::from_be_bytes(*b"inac") as _);
68}
