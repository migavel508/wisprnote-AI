84#[cfg(test)]
85mod tests {
86    use crate::{ns, sc};
87
88    #[tokio::test]
89    async fn basics() {
90        let content = sc::ShareableContent::current_process().await.unwrap();
91        let displays = content.displays();
92        let display = displays.first().unwrap();
93        let filter = sc::ContentFilter::with_display_excluding_windows(&display, &ns::Array::new());
94        let cfg = sc::StreamCfg::new();
95        let buf = sc::ScreenshotManager::capture_sample_buf(&filter, &cfg)
96            .await
97            .unwrap();
98        let image_buf = buf.image_buf().unwrap();
99        assert!(image_buf.width() > 0);
100        assert!(image_buf.height() > 0);
101
102        let image = sc::ScreenshotManager::capture_image(&filter, &cfg)
103            .await
104            .unwrap();
105        assert!(image.width() > 0);
106        assert!(image.height() > 0);
107    }
