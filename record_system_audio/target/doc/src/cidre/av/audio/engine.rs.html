27#[derive(PartialEq, Eq, Debug, Clone, Copy)]
28#[repr(isize)]
29pub enum ManualRenderingStatus {
30    /// An error occurred when rendering and no data was returned. See the returned error code
31    /// for the description of the error.
32    Error = -1,
33    /// All of the requested data was returned successfully.
34    Success = 0,
35    /// Applicable only to the input node, when it provides input data for rendering
36    /// (see `AVAudioInputNode(setManualRenderingInputPCMFormat:inputBlock:)`).
37    /// Indicates that not enough input data was returned by the input node to satisfy the
38    /// render request at the current time. The output buffer may contain data rendered by other
39    /// active sources in the engine's processing graph.
40    InsufficientDataFromInputNode = 1,
41    /// The operation could not be performed now, but the client could retry later if needed.
42    /// This is usually to guard a realtime render operation (e.g. rendering through
43    /// `manualRenderingBlock`) when a reconfiguration of the engine's internal state
44    /// is in progress.
45    CannotDoInCurrentContext = 2,
46}

48/// By default, the engine is connected to an audio device and automatically renders in realtime.
49/// It can also be configured to operate in manual rendering mode, i.e. not connected to an
50/// audio device and rendering in response to requests from the client.
51#[derive(PartialEq, Eq, Debug, Clone, Copy)]
52#[repr(isize)]
53pub enum ManualRenderingMode {
54    /// The engine operates in an offline mode without any realtime constraints.
55    Offline = 0,
56    /// The engine operates under realtime constraints, i.e. it will not make any blocking call
57    /// (e.g. calling libdispatch, blocking on a mutex, allocating memory etc.) while rendering.
58    /// Note that only the block based render mechanism can be used in this mode
59    /// (see `AVAudioEngine(manualRenderingBlock)`.
60    Realtime = 1,
61}

68/// An AVAudioEngine contains a group of connected AVAudioNodes ("nodes"), each of which performs
69/// an audio signal generation, processing, or input/output task.
70///
71/// Nodes are created separately and attached to the engine.
72///
73/// The engine supports dynamic connection, disconnection and removal of nodes while running,
74/// with only minor limitations:
75/// - all dynamic reconnections must occur upstream of a mixer
76/// - while removals of effects will normally result in the automatic connection of the adjacent
77///   nodes, removal of a node which has differing input vs. output channel counts, or which
78///   is a mixer, is likely to result in a broken graph.
79///
80/// By default, the engine is connected to an audio device and automatically renders in realtime.
81/// It can also be configured to operate in manual rendering mode, i.e. not connected to an
82/// audio device and rendering in response to requests from the client, normally at or
83/// faster than realtime rate.

200    /// Set the engine to operate in a manual rendering mode with the specified render format and
201    /// maximum frame count.
202    ///
203    /// Use this method to configure the engine to render in response to requests from the client.
204    /// The engine must be in a stopped state before calling this method.
205    /// The render format must be a PCM format and match the format of the buffer to which
206    /// the engine is asked to render (see `renderOffline:toBuffer:error:`).
207    ///
208    /// It is advised to enable manual rendering mode soon after the engine is created, and
209    /// before accessing any of mainMixerNode, inputNode or outputNode of the engine.
210    /// Otherwise, accessing or interacting with the engine before enabling manual rendering
211    /// mode could have the unintended side-effect of configuring the hardware for device-rendering
212    /// mode.
213    ///
214    /// The input data in manual rendering mode can be supplied through the source nodes, e.g.
215    /// `av::audio::PlayerNode`, `av::audio::InputNode` etc.
216    ///
217    /// When switching to manual rendering mode, the engine:
218    /// 1. Switches the input and output nodes to manual rendering mode. Their input and output
219    ///     formats may change.
220    /// 2. Removes any taps previously installed on the input and output nodes.
221    /// 3. Maintains all the engine connections as is.
222    #[inline]
223    pub fn enable_manual_rendering_mode<'ear>(
224        &mut self,
225        mode: ManualRenderingMode,
226        format: &av::AudioFormat,
227        max_frame_count: av::AudioFrameCount,
228    ) -> ns::Result<'ear> {
229        ns::if_err(|err| unsafe {
230            self.enable_manual_rendering_mode_err(mode, format, max_frame_count, err)
231        })
232    }

235    pub fn disable_manual_rendering_mode(&mut self);

237    #[objc::msg_send(renderOffline:toBuffer:error:)]
238    pub fn render_offline_err<'ar>(
239        &mut self,
240        number_of_frames: av::AudioFrameCount,
241        to_buf: &mut av::audio::PcmBuf,
242        error: *mut Option<&'ar ns::Error>,
243    ) -> av::audio::EngineManualRenderingStatus;
244
245    #[inline]
246    pub fn render_offline<'ar>(
247        &mut self,
248        number_of_frames: av::AudioFrameCount,
249        to_buf: &mut av::audio::PcmBuf,
250    ) -> Result<av::audio::EngineManualRenderingStatus, &'ar ns::Error> {
251        let mut error = None;
252        let status = self.render_offline_err(number_of_frames, to_buf, &mut error);
253        if status == ManualRenderingStatus::Error && error.is_some() {
254            return Err(unsafe { error.unwrap_unchecked() });
255        }
256        Ok(status)
257    }

329    fn manual_redering_modes() {
330        let format =
331            av::audio::Format::standard_with_sample_rate_and_channels(44_100.0, 2).unwrap();
332        let mut engine = av::audio::Engine::new();
333        engine
334            .enable_manual_rendering_mode(
335                av::audio::engine::ManualRenderingMode::Offline,
336                &format,
337                1024,
338            )
339            .expect("Failed to enter manual rendering mode");
340        assert!(engine.is_in_manual_rendering_mode());
341
342        assert_eq!(
343            engine.manual_rendering_format().channel_count(),
344            format.channel_count()
345        );
346
347        let mut pcm_buf = av::audio::PcmBuf::with_format(&format, 1024).unwrap();
348
349        pcm_buf.set_frame_len(1024).unwrap();
350
351        engine.start().expect("Failed to start engine");
352        assert!(engine.is_running());
353
354        // this is segfault with swift too https://gist.github.com/yury/2a74943c65b69a3a593a2c096ac36b54
355        // we didn't connect any inputs :)
356        // let status = engine.render_offline_err(1024, &mut pcm_buf, std::ptr::null_mut());
357
358        engine.stop();
359
360        let player_node = av::AudioPlayerNode::new();
361        engine.attach_node(&player_node);
362        let mixer_node = engine.main_mixer_node().retained();
363        engine.connect_node_to_node(&player_node, &mixer_node, None);
364        engine.start().expect("Failed to start engine");
365        assert_eq!(engine.manual_rendering_sample_time(), 0);
366        player_node.play();
367
368        let status = engine.render_offline(1024, &mut pcm_buf).unwrap();

370        assert_eq!(status, av::audio::EngineManualRenderingStatus::Success);
371        assert_eq!(pcm_buf.frame_len(), 1024);
372        assert_eq!(engine.manual_rendering_sample_time(), 1024);
