5/// Filter types available to use with EQ.
6#[repr(isize)]
7pub enum FilterType {
8    /// Parametric filter based on Butterworth analog prototype.
9    /// Required parameters: frequency (center), bandwidth, gain
10    Parametric = 0,
11
12    /// Simple Butterworth 2nd order low pass filter
13    /// Required parameters: frequency (-3 dB cutoff at specified frequency)
14    LowPass = 1,
15
16    /// Simple Butterworth 2nd order high pass filter
17    /// Required parameters: frequency (-3 dB cutoff at specified frequency)
18    HighPass = 2,
19
20    /// Low pass filter with resonance support (via bandwidth parameter)
21    ///  Required parameters: frequency (-3 dB cutoff at specified frequency), bandwidth
22    ResonantLowPass = 3,
23
24    /// High pass filter with resonance support (via bandwidth parameter)
25    /// Required parameters: frequency (-3 dB cutoff at specified frequency), bandwidth
26    ResonantHighPass = 4,
27
28    /// Band pass filter
29    /// Required parameters: frequency (center), bandwidth
30    BandPass = 5,
31
32    /// Band stop filter (aka "notch filter")
33    /// Required parameters: frequency (center), bandwidth
34    BandStop = 6,
35
36    /// Low shelf filter
37    /// Required parameters: frequency (center), gain
38    LowShelf = 7,
39
40    /// High shelf filter
41    /// Required parameters: frequency (center), gain
42    HighShelf = 8,
43
44    /// Low shelf filter with resonance support (via bandwidth parameter)
45    /// Required parameters: frequency (center), bandwidth, gain
46    ResonantLowShelf = 9,
47
48    /// High shelf filter with resonance support (via bandwidth parameter)
49    /// Required parameters: frequency (center), bandwidth, gain
50    ResonantHighShelf = 10,
51}
