148fn pcm_slice_at<T>(buf: &PcmBuf, ptr: *const *const T, index: usize) -> Option<&[T]> {
149    if ptr.is_null() {
150        return None;
151    }
152
153    let len = buf.frame_len() as usize;
154    let stride = buf.stride() as usize;
155    if stride > 1 {
156        // if stride is not 1 it is interleaved, single buf
157        // save some calls to format
158        let slice = unsafe { std::slice::from_raw_parts(ptr, 1) };
159        let data = unsafe { std::slice::from_raw_parts(slice[0], stride * len) };
160        return Some(data);
161    }
162    let channels_n = buf.format().channel_count();
163    let slice = unsafe { std::slice::from_raw_parts(ptr, channels_n as _) };
164    let data = unsafe { std::slice::from_raw_parts(slice[index], len) };
165    Some(data)
166}
167
168fn pcm_slice_at_mut<T>(buf: &mut PcmBuf, ptr: *const *mut T, index: usize) -> Option<&mut [T]> {
169    if ptr.is_null() {
170        return None;
171    }
172
173    let len = buf.frame_len() as usize;
174    let stride = buf.stride() as usize;
175    if stride > 1 {
176        // interleaved, single buf
177        let slice = unsafe { std::slice::from_raw_parts(ptr, 1) };
178        let data = unsafe { std::slice::from_raw_parts_mut(slice[0], stride * len) };
179        return Some(data);
180    }
181    let channels_n = buf.format().channel_count();
182    let slice = unsafe { std::slice::from_raw_parts(ptr, channels_n as _) };
183    let data = unsafe { std::slice::from_raw_parts_mut(slice[index], len) };
184    Some(data)
185}

285    #[test]
286    fn basics() {
287        let channel_count = 2;
288        let format =
289            av::AudioFormat::standard_with_sample_rate_and_channels(44800.0, channel_count)
290                .unwrap();
291        let cap = 1024;
292        let mut buf = av::AudioPcmBuf::with_format(&format, cap).unwrap();
293        buf.set_frame_len(cap).unwrap();
294        let data = buf.data_f32();
295        let (n, cap) = if format.is_interleaved() {
296            (1, channel_count * cap)
297        } else {
298            (channel_count, cap)
299        };
300        let buf = unsafe { std::slice::from_raw_parts(data, n as _) };
301        let channel = buf[0];
302        let left = unsafe { std::slice::from_raw_parts(channel, cap as _) };
303        let sum = left.iter().sum();
304        assert_eq!(0.0f32, sum);
305    }
