227define_obj_type!(
228    #[doc(alias = "AVMetadataObject")]
229    pub Obj(ns::Id)
230);
231
232impl Obj {
233    /// The value of this property is a cg::Rect representing the bounding rectangle
234    /// of the object with respect to the picture in which it resides. The rectangle's
235    /// origin is top left. If the metadata originates from video, bounds may be
236    /// expressed as scalar values from 0. - 1. If the original video has been scaled
237    /// down, the bounds of the metadata object still are meaningful. This property may
238    /// return cg::Rect::zero if the metadata has no bounds.
239    #[objc::msg_send(bounds)]
240    pub fn bounds(&self) -> cg::Rect;
241
242    /// The media time associated with this metadata object.
243    ///
244    /// The value of this property is a cm::Time associated with the metadata object.
245    /// For capture, it is the time at which this object was captured. If this metadata
246    /// object originates from a cm::SampleBuffer, its time matches the sample buffer's
247    /// presentation time. This property may return cm::Time::invalid.
248    #[cfg(feature = "cm")]
249    #[objc::msg_send(time)]
250    pub fn time(&self) -> cm::Time;
251
252    /// The media duration associated with this metadata object.
253    ///
254    /// The value of this property is a cm::Time representing the duration
255    /// of the metadata object. If this metadata object originates from a cm::SampleBuffer,
256    /// its duration matches the sample buffer's duration.
257    /// This property may return cm::Time::invalid.
258    #[cfg(feature = "cm")]
259    #[objc::msg_send(duration)]
260    pub fn duration(&self) -> cm::Time;
261
262    /// An identifier for the metadata object.
263    ///
264    /// Clients inspecting a collection of metadata objects can use this property to filter objects
265    /// with a matching type.
266    #[objc::msg_send(type)]
267    pub fn obj_type(&self) -> &Type;
268
269    /// A number associated with object groups (e.g., face and body) that is unique for each physical object
270    /// (e.g., a person whom the face and body belong to).
271    ///
272    /// The value of this property is an NSInteger indicating the unique identifier to combine objects (for instance,
273    /// face and body) into groups (a physical person). A human body and face for the same person will have the same group ID.
274    /// It is set to -1 when it's invalid or not available. When it's set to a value of >=0, it is unique across all object groups.
275    #[objc::msg_send(groupID)]
276    #[api::available(macos = 26.0, ios = 26.0, maccatalyst = 26.0, tvos = 26.0)]
277    pub fn group_id(&self) -> isize;
278
279    /// A unique identifier for each detected object type (face, body, hands, heads and salient objects).
280    ///
281    /// Defaults to a value of -1 when it is invalid or not available. When a new object enters the picture,
282    /// it is assigned a new unique identifier. obj_idss are not re-used as objects leave the picture and new ones enter.
283    ///  Objects that leave the picture then re-enter are assigned a new objectID.
284    #[objc::msg_send(objectID)]
285    #[api::available(macos = 26.0, ios = 26.0, maccatalyst = 26.0, tvos = 26.0)]
286    pub fn obj_id(&self) -> isize;
287}

358impl FaceObj {
359    /// A unique number associated with the receiver.
360    ///
361    /// The value of this property is an isize indicating the unique identifier of this face in the picture.
362    /// When a new face enters the picture, it is assigned a new unique identifier. face_ids are not re-used as faces
363    /// leave the picture and new ones enter. Faces that leave the picture then re-enter are assigned a new face_id.
364    #[objc::msg_send(faceID)]
365    pub fn face_id(&self) -> isize;
366
367    /// The roll angle of the face in degrees.
368    ///
369    /// The value of this property is a cg::Float indicating the face's angle of roll (or tilt) in degrees.
370    /// A value of 0.0 indicates that the face is level in the picture. If -has_roll_angle returns false, then reading this property throws
371    /// an ns::GenericException.
372    #[objc::msg_send(hasRollAngle)]
373    pub fn has_roll_angle(&self) -> bool;
374
375    #[objc::msg_send(rollAngle)]
376    pub unsafe fn roll_angle_throws(&self) -> cg::Float;
377
378    /// The roll angle of the face in degrees.
379    #[inline]
380    pub fn roll_angle(&self) -> Option<cg::Float> {
381        if self.has_roll_angle() {
382            Some(unsafe { self.roll_angle_throws() })
383        } else {
384            None
385        }
386    }
387    #[objc::msg_send(hasYawAngle)]
388    pub fn has_yaw_angle(&self) -> bool;
389
390    /// The yaw angle of the face in degrees.
391    ///
392    /// The value of this property is a cg::Float indicating the face's angle of yaw (or turn) in degrees.
393    /// A value of 0.0 indicates that the face is straight on in the picture.
394    /// If has_yaw_angle() returns false, then reading this property throws an ns::GenericException.
395    #[objc::msg_send(yawAngle)]
396    pub unsafe fn yaw_angle_throws(&self) -> cg::Float;
397
398    /// The yaw angle of the face in degrees.
399    #[inline]
400    pub fn yaw_angle(&self) -> Option<cg::Float> {
401        if self.has_yaw_angle() {
402            Some(unsafe { self.yaw_angle_throws() })
403        } else {
404            None
405        }
406    }
407}
