8#[doc(alias = "AVAssetWriterStatus")]
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10#[repr(isize)]
11pub enum Status {
12    /// Indicates that the status of the asset writer is not currently known.
13    Unknown = 0,
14    /// Indicates that the asset writer is successfully writing samples to its output file.
15    Writing = 1,
16    /// Indicates that the asset writer has successfully written all samples following a call to finishWriting.
17    Completed = 2,
18    /// Indicates that the asset writer can no longer write samples to its output file because of an error. The error is described by the value of the asset writer's error property.
19    Failed = 3,
20    /// Indicates that the asset writer can no longer write samples because writing was canceled with the cancelWriting method.
21    Cancelled = 4,
22}
