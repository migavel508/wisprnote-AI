40/// Is an abstract class that defines an interface for reading a single collection
41/// of samples of a common media type from an av::AssetReader.
42///
43/// Clients can read the media data of an asset by adding one or more concrete
44/// instances of av::AssetReaderOutput to an AVAssetReader using the -[AVAssetReader addOutput:] method.
45///
46/// IMPORTANT PERFORMANCE NOTE: Make sure to set the always_copies_sample_data property to false
47/// if you do not need to modify the sample data in-place, to avoid unnecessary and inefficient copying.
48impl ReaderOutput {
49    #[objc::msg_send(mediaType)]
50    pub fn media_type(&self) -> arc::R<av::MediaType>;
51
52    /// Indicates whether or not the data in buffers gets copied before being vended to the client.
53    ///
54    /// When the value of this property is true, the AVAssetReaderOutput will always vend a buffer
55    /// with copied data to the client. Data in such buffers can be freely modified by the client.
56    /// When the value of this property is false, the buffers vended to the client may not be copied.
57    /// Such buffers may still be referenced by other entities. The result of modifying a buffer
58    /// whose data hasn't been copied is undefined. Requesting buffers whose data hasn't been copied
59    /// when possible can lead to performance improvements.
60    ///
61    /// Default value is true
62    #[objc::msg_send(alwaysCopiesSampleData)]
63    pub fn always_copies_sample_data(&self) -> bool;
64
65    #[objc::msg_send(setAlwaysCopiesSampleData:)]
66    pub fn set_always_copies_sample_data(&mut self, value: bool);
67
68    #[objc::msg_send(supportsRandomAccess)]
69    pub fn supports_random_access(&self) -> bool;
70
71    /// Copies the next sample buffer for the output synchronously.
72    ///
73    /// This method will return None if there are no more sample buffers available for
74    /// the receiver within the time range specified by its av::AssetReader's time_range property,
75    /// or if there is an error that prevents the av::AssetReader from reading more media data.
76    /// When this method returns None, clients should check the value of the associated av::AssetReader's
77    /// status property to determine why no more samples could be read.
78    ///
79    /// This method throws an exception if this output is not added to an instance of av::AssetReader
80    /// (using add_output()) and start_reading() is not called on that asset reader.
81    #[cfg(feature = "cm")]
82    #[objc::msg_send(copyNextSampleBuffer)]
83    pub unsafe fn next_sample_buf_throws(&mut self) -> Option<arc::Retained<cm::SampleBuf>>;
84
85    #[cfg(feature = "cm")]
86    #[inline]
87    pub fn next_sample_buf<'ear>(&mut self) -> ns::ExResult<'ear, Option<arc::R<cm::SampleBuf>>> {
88        ns::try_catch(|| unsafe { self.next_sample_buf_throws() })
89    }
90}

104    /// Returns an instance of AVAssetReaderTrackOutput for reading from the specified track and
105    /// supplying media data according to the specified output settings.
106    ///
107    /// The track must be one of the tracks contained by the target AVAssetReader's asset.
108    /// A value of nil for outputSettings configures the output to vend samples in their
109    /// original format as stored by the specified track.  Initialization will fail if the output settings
110    /// cannot be used with the specified track.
111    ///
112    /// ReaderTrackOutput can only produce uncompressed output. For audio output settings, this means that
113    /// AVFormatIDKey must be kAudioFormatLinearPCM. For video output settings, this means that the dictionary
114    /// must follow the rules for uncompressed video output, as laid out in AVVideoSettings.h.
115    /// ReaderTrackOutput does not support the AVAudioSettings.h key AVSampleRateConverterAudioQualityKey
116    /// or the following AVVideoSettings.h keys:
117    ///
118    ///  - AVVideoCleanApertureKey
119    ///  - AVVideoPixelAspectRatioKey
120    ///  - AVVideoScalingModeKey
121    ///
122    /// When constructing video output settings the choice of pixel format will affect the performance
123    /// and quality of the decompression. For optimal performance when decompressing video the requested pixel
124    /// format should be one that the decoder supports natively to avoid unnecessary conversions.
125    /// Below are some recommendations:
126    ///   - For H.264 use kCVPixelFormatType_420YpCbCr8BiPlanarVideoRange, or kCVPixelFormatType_420YpCbCr8BiPlanarFullRange
127    ///     if the video is known to be full range. For JPEG on iOS, use kCVPixelFormatType_420YpCbCr8BiPlanarFullRange.
128    ///   - For other codecs on OSX, kCVPixelFormatType_422YpCbCr8 is the preferred pixel format for video
129    ///     and is generally the most performant when decoding. If you need to work in the RGB domain then kCVPixelFormatType_32BGRA
130    ///     is recommended on iOS and kCVPixelFormatType_32ARGB is recommended on OSX.
131    ///   - ProRes encoded media can contain up to 12bits/ch. If your source is ProRes encoded and you wish to preserve more
132    ///     than 8bits/ch during decompression then use one of the following pixel formats:
133    ///     kCVPixelFormatType_4444AYpCbCr16, kCVPixelFormatType_422YpCbCr16, kCVPixelFormatType_422YpCbCr10, or kCVPixelFormatType_64ARGB.
134    ///     av::AssetReader does not support scaling with any of these high bit depth pixel formats.
135    ///     If you use them then do not specify kCVPixelBufferWidthKey or kCVPixelBufferHeightKey in your outputSettings dictionary.
136    ///     If you plan to append these sample buffers to an AVAssetWriterInput then note that only the ProRes encoders support these pixel formats.
137    ///   - ProRes 4444 encoded media can contain a mathematically lossless alpha channel. To preserve the alpha channel during decompression use
138    ///     a pixel format with an alpha component such as kCVPixelFormatType_4444AYpCbCr16 or kCVPixelFormatType_64ARGB.
139    ///     To test whether your source contains an alpha channel check that the track's format description has kCMFormatDescriptionExtension_Depth
140    ///     and that its value is 32.
141    pub fn with_track_throws(
142        track: &av::asset::Track,
143        output_settings: Option<&ns::Dictionary<ns::String, ns::Id>>,
144    ) -> arc::R<Self> {
145        Self::alloc().init_with_track_throws(track, output_settings)
146    }
147

151    ) -> ns::ExResult<'ear, arc::R<Self>> {
152        ns::try_catch(|| Self::with_track_throws(track, output_settings))
153    }
