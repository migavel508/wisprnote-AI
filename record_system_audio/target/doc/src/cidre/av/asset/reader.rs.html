6#[doc(alias = "AVAssetReaderStatus")]
7#[derive(Debug, Clone, Copy, PartialEq, Eq)]
8#[repr(isize)]
9pub enum Status {
10    /// Indicates that the status of the asset reader is not currently known.
11    Unknown = 0,
12    /// Indicates that the asset reader is successfully reading samples from its asset.
13    Reading = 1,
14    /// Indicates that the asset reader has successfully read all of the samples in its time range.
15    Completed = 2,
16    /// Indicates that the asset reader can no longer read samples from its asset because of an error. The error is described by the value of the asset reader's error property.
17    Failed = 3,
18    /// Indicates that the asset reader can no longer read samples because reading was canceled with the cancelReading method.
19    Cancelled = 4,
20}
