31/// States progress forward and do not move backwards.
32#[derive(Debug, Eq, PartialEq, Copy, Clone)]
33#[doc(alias = "nw_listener_state_t")]
34#[repr(i32)]
35pub enum State {
36    /// The state of the listener is not valid. This state
37    /// will never be delivered in the listener's state update handler, and can be treated as
38    /// an unexpected value.
39    Invalid = 0,
40
41    /// The listener is waiting for a usable network before being able to receive connections
42    Waiting = 1,
43
44    /// The listener is ready and able to accept incoming connections
45    Ready = 2,
46
47    /// The listener has irrecoverably closed or failed
48    Failed = 3,
49
50    /// The listener has been cancelled by the caller
51    Cancelled = 4,
52}

236#[cfg(test)]
237mod tests {
238    use crate::{dispatch, nw};
239
240    #[test]
241    fn basics() {
242        let sema = dispatch::Semaphore::new(0);
243        let q = dispatch::Queue::new();
244        let mut listener = nw::Listener::with_port(c"9089", &nw::Params::app_service()).unwrap();
245        let mut sema_guard = Some(sema.guard());
246        let block_listener = listener.retained();
247
248        listener.set_state_changed_handler(move |state, err| {
249            assert!(err.is_none());
250            assert_eq!(state, nw::ListenerState::Ready);
251            assert_eq!(block_listener.port(), 9089);
252
253            sema_guard.take();
254        });
255
256        listener.set_new_connection_handler(|_conn| {});
257
258        assert_eq!(listener.port(), 0);
259
260        listener.start(&q);
261
262        sema.wait_forever();
263    }
