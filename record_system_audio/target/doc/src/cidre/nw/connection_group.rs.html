23#[doc(alias = "nw_connection_group_state_t")]
24#[derive(Debug, Copy, Clone, Eq, PartialEq)]
25#[repr(i32)]
26pub enum State {
27    /// The state of the connection group is not valid.
28    /// This state will never be delivered in the connection group's state update
29    /// handler and can be treated as an unexpected value.
30    #[doc(alias = "nw_connection_group_state_invalid")]
31    Invalid = 0,
32
33    /// The connection group is waiting for a usable network
34    /// before being able to receive and process incoming messages.
35    #[doc(alias = "nw_connection_group_state_waiting")]
36    Waiting = 1,
37
38    /// The connection group is ready and able to receive and
39    /// process incoming messages.
40    #[doc(alias = "nw_connection_group_state_ready")]
41    Ready = 2,
42
43    /// The connection group has irrecoverably failed.
44    /// You should cancel the connection group and create a new connection group object
45    /// if you wish to continue processing incoming messages.
46    #[doc(alias = "nw_connection_group_state_failed")]
47    Failed = 3,
48
49    /// The connection group has been cancelled by
50    /// the user. You should create a new connection group object if you wish to continue
51    /// processing incoming messages.
52    #[doc(alias = "nw_connection_group_state_cancelled")]
53    Cancelled = 4,
54}
