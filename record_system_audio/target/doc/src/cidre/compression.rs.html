68const COMPRESSION_STREAM_FINALIZE: i32 = 0x0001;
69
70#[doc(alias = "compression_stream_init")]
71#[doc(alias = "compression_stream_process")]
72#[doc(alias = "compression_stream_destroy")]
73pub fn stream(op: StreamOp, src: &[u8], algorithm: Algorithm) -> Result<Vec<u8>, Status> {
74    let size = if op == StreamOp::Encode {
75        src.len() / 4
76    } else {
77        src.len() * 4
78    };
79    let mut dst = vec![0u8; size.max(1)];
80    let mut stream = Stream {
81        dst_ptr: dst.as_mut_ptr(),
82        dst_size: dst.len(),
83        src_ptr: src.as_ptr(),
84        src_size: src.len(),
85        state: std::ptr::null_mut(),
86    };
87
88    unsafe {
89        let res = compression_stream_init(&mut stream, op, algorithm);
90        if res != Status::Ok {
91            return Err(res);
92        }
93        loop {
94            let res = compression_stream_process(&mut stream, COMPRESSION_STREAM_FINALIZE);
95            if res == Status::Ok {
96                let len = dst.len();
97                dst.resize((dst.len() * 2).max(512), 0u8);
98                stream.dst_ptr = dst.as_mut_ptr().add(len);
99                stream.dst_size = dst.len() - len;
100                continue;
101            }
102            let r = compression_stream_destroy(&mut stream);
103            if r != Status::Ok {
104                break Err(r);
105            }
106            break if res == Status::End {
107                dst.truncate(dst.len() - stream.dst_size);
108                Ok(dst)
109            } else {
110                Err(res)
111            };
112        }
113    }
114}

164#[cfg(test)]
165mod tests {
166    use crate::compression::*;
167    #[test]
168    fn basics() {
169        let len = encode_scratch_buf_len(Algorithm::ZLIB);
170        assert_eq!(270336, len);
171
172        let len = encode_scratch_buf_len(Algorithm(10));
173        assert_eq!(0, len);
174
175        let mut encoded_buf = vec![0u8; 100];
176
177        let data = "hello compression!!!!";
178
179        let size = encode_buf(&mut encoded_buf, data.as_bytes(), Algorithm::LZMA);
180        assert_eq!(size, 72);
181
182        let mut decoded_buf = vec![0u8; 100];
183        let size = decode_buf(&mut decoded_buf, &encoded_buf[..size], Algorithm::LZMA);
184
185        assert_eq!(size, data.len());
186        assert_eq!(data.as_bytes(), &decoded_buf[..size]);
187
188        let decoded_buf = stream(StreamOp::Decode, &encoded_buf, Algorithm::LZMA).unwrap();
189        assert_eq!(&decoded_buf, data.as_bytes());
190
191        let encoded_buf = stream(StreamOp::Encode, data.as_bytes(), Algorithm::LZMA).unwrap();
192        assert_eq!(encoded_buf.len(), 72);
193
194        let decoded_buf = stream(StreamOp::Decode, &encoded_buf, Algorithm::LZMA).unwrap();
195        assert_eq!(&decoded_buf, data.as_bytes());
196    }
