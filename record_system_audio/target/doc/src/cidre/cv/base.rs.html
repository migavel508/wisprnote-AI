105define_opts!(
106    #[doc(alias = "CVTimeStampFlags")]
107    pub TimeStampFlags(u64)
108);
109
110impl TimeStampFlags {
111    pub const VIDEO_TIME_VALID: Self = Self(1 << 0);
112    pub const HOST_TIME_VALID: Self = Self(1 << 1);
113    pub const SMPTE_TIME_VALID: Self = Self(1 << 2);
114    pub const VIDEO_REFRESH_PERIOD_VALID: Self = Self(1 << 3);
115    pub const RATE_SCALAR_VALID: Self = Self(1 << 4);
116
117    // There are flags for each field to make it easier to detect interlaced vs progressive output
118    pub const TOP_FIELD: Self = Self(1 << 16);
119    pub const BOTTOM_FIELD: Self = Self(1 << 17);
120
121    // Some commonly used combinations of timestamp flags
122    pub const VIDEO_HOST_TIME_VALID: Self =
123        Self(Self::VIDEO_TIME_VALID.0 | Self::HOST_TIME_VALID.0);
124    pub const IS_INTERLACED: Self = Self(Self::TOP_FIELD.0 | Self::BOTTOM_FIELD.0);
125}
