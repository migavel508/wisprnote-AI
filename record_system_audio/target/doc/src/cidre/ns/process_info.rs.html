3#[doc(alias = "NSProcessInfoThermalState")]
4#[derive(PartialEq, Eq, Clone, Copy, Debug)]
5#[repr(isize)]
6pub enum ThermalState {
7    /// No corrective action is needed.
8    Nominal,
9
10    /// The system has reached a state where fans may become audible
11    /// (on systems which have fans). Recommendation: Defer non-user-visible activity.
12    Fair,
13
14    /// Fans are running at maximum speed (on systems which have fans), system performance
15    /// may be impacted. Recommendation: reduce application's usage of CPU, GPU and I/O,
16    /// if possible. Switch to lower quality visual effects, reduce frame rates.
17    Serious,
18
19    /// System performance is significantly impacted and the system needs to cool down.
20    /// Recommendation: reduce application's usage of CPU, GPU, and I/O to the minimum
21    /// level needed to respond to user actions. Consider stopping use of camera and
22    /// other peripherals if your application is using them.
23    Critical,
24}

133    #[test]
134    fn basics() {
135        let pi = ns::ProcessInfo::current();
136
137        assert_ne!(pi.thermal_state(), ns::ProcessInfoThermalState::Critical);
138        assert_eq!(pi.is_low_power_mode_enabled(), false);
139        assert!(pi.processor_count() > 1);
140        assert!(pi.active_processor_count() > 1);
141        assert!(!pi.is_ios_app_on_mac());
142        assert!(!pi.is_mac_catalyst_app());
143    }
