190impl AttrString {
191    /// For performance reasons, this property returns the current backing store of
192    /// the attributed string object. If you want to maintain a snapshot of this as
193    /// you manipulate the returned string, you should make a copy of the appropriate substring.
194    #[objc::msg_send(string)]
195    pub fn string(&self) -> arc::R<ns::String>;
196
197    #[objc::msg_send(length)]
198    pub fn len(&self) -> usize;
199
200    #[inline]
201    pub fn is_empty(&self) -> bool {
202        self.len() == 0
203    }
204
205    #[objc::msg_send(attributesAtIndex:effectiveRange:)]
206    pub unsafe fn attrs_at_throws(
207        &self,
208        index: usize,
209        effective_range: Option<&ns::Range>,
210    ) -> arc::R<ns::Dictionary<AttrStringKey, ns::Id>>;
211
212    pub fn attrs_at<'ear>(
213        &self,
214        index: usize,
215        effective_range: Option<&ns::Range>,
216    ) -> ns::ExResult<'ear, arc::R<ns::Dictionary<AttrStringKey, ns::Id>>> {
217        ns::try_catch(|| unsafe { self.attrs_at_throws(index, effective_range) })
218    }
219
220    #[inline]
221    pub fn with_str(str: &str) -> arc::R<Self> {
222        let str = ns::String::with_str(str);

224    }

368#[cfg(all(test, target_os = "macos"))]
369mod tests {
370    use crate::{ns, objc::Obj};
371
372    #[test]
373    fn basics() {
374        let str = ns::String::with_str("Hello");
375        let astr = ns::AttrString::with_string(&str);
376
377        assert_eq!(&astr.string(), &str);
378        let attrs = astr.attrs_at(0, None).unwrap();
379        assert!(attrs.is_empty());
380
381        let copy = astr.copy_mut();
382        let copy = copy.copy();
383        let copy = ns::AttrString::with_attr_string(&copy);
384
385        assert_eq!(copy.len(), 5);
386
387        let mut mcopy = copy.copy_mut();
388
389        // let mut mstr = mcopy.string_mut();
390
391        let color_key = ns::AttrStringKey::foreground_color();
392        let color_val = ns::Color::with_rgba(0.5, 0.5, 1.0, 1.0);
393
394        let attrs = ns::Dictionary::with_keys_values(&[color_key], &[color_val.as_id_ref()]);
395
396        mcopy.add_attrs(&attrs, ns::Range::new(0, 5)).expect("ok");
397        mcopy
398            .add_attr(color_key, &color_val, ns::Range::new(1000, 10))
399            .expect_err("ok");
400        mcopy
401            .add_attr(color_key, &color_val, ns::Range::new(0, 3))
402            .expect("ok");
403
404        mcopy
405            .remove_attr(color_key, ns::Range::new(100, 200))
406            .expect_err("should be out of bounds");
407
408        mcopy
409            .remove_attr(color_key, ns::Range::new(0, 5))
410            .expect("failed to remove valid range");
411
412        let mstr = ns::AttrStringMut::with_str("nice");
413        assert_eq!(mstr.len(), 4);
414        // TODO: investigate

416        // .expect_err("Should be Exception");
417    }
