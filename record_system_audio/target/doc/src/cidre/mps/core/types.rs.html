8/// Options used when creating mps::Kernel objects
9impl KernelOpts {
10    /// Use default options
11    #[doc(alias = "MPSKernelOptionsNone")]
12    pub const NONE: Self = Self(0usize);
13
14    /// Most MPS functions will sanity check their arguments. This has a small but
15    /// non-zero CPU cost. Setting the MPSKernelOptionsSkipAPIValidation will skip these checks.
16    /// MPSKernelOptionsSkipAPIValidation does not skip checks for memory allocation failure.
17    /// Caution:  turning on MPSKernelOptionsSkipAPIValidation can result in undefined behavior
18    /// if the requested operation can not be completed for some reason. Most error states
19    /// will be passed through to Metal which may do nothing or abort the program if Metal
20    /// API validation is turned on.
21    #[doc(alias = "MPSKernelOptionsSkipAPIValidation")]
22    pub const SKIP_APIVALIDATION: Self = Self(1usize << 0);
23
24    /// When possible, MPSKernels use a higher precision data representation internally than
25    /// the destination storage format to avoid excessive accumulation of computational
26    /// rounding error in the result. MPSKernelOptionsAllowReducedPrecision advises the
27    /// MPSKernel that the destination storage format already has too much precision for
28    /// what is ultimately required downstream, and the MPSKernel may use reduced precision
29    /// internally when it feels that a less precise result would yield better performance.
30    /// The expected performance win is often small, perhaps 0-20%. When enabled, the
31    /// precision of the result may vary by hardware and operating syste
32    #[doc(alias = "MPSKernelOptionsAllowReducedPrecision")]
33    pub const ALLOW_REDUCED_PRECISION: Self = Self(1usize << 1);
34
35    /// Some MPSKernels may automatically split up the work internally into multiple tiles.
36    /// This improves performance on larger textures and reduces the amount of memory needed by
37    /// MPS for temporary storage. However, if you are using your own tiling scheme to achieve
38    /// similar results, your tile sizes and MPS's choice of tile sizes may interfere with
39    /// one another causing MPS to subdivide your tiles for its own use inefficiently. Pass
40    /// MPSKernelOptionsDisableInternalTiling to force MPS to process your data tile as a
41    /// single chunk.
42    #[doc(alias = "MPSKernelOptionsDisableInternalTiling")]
43    pub const DISABLE_INTERNAL_TILING: Self = Self(1usize << 2);
44
45    /// Enabling this bit will cause various -encode... methods to call MTLCommandEncoder
46    /// push/popDebugGroup.  The debug string will be drawn from MPSKernel.label, if any
47    /// or the name of the class otherwise.
48    #[doc(alias = "MPSKernelOptionsInsertDebugGroups")]
49    pub const INSERT_DEBUG_GROUPS: Self = Self(1usize << 3);
50
51    /// Some parts of MPS can provide debug commentary and tuning advice when run.
52    /// Setting this bit to 1 will cause the commentary to be emitted to stderr. Otherwise,
53    /// the code is silent.  This is especially useful for debugging MPSNNGraph. This option
54    /// is on by default when the MPS_LOG_INFO environment variable is defined.  For
55    /// even more detailed output on a MPS object, you can use the po command in llvm
56    /// with MPS objects:
57    /// ```llvm
58    /// po <mps pointer>
59    /// ```
60    pub const VERBOSE: Self = Self(1usize << 4);
61}

63/// Options used to control edge behaviour of filter when filter reads beyond boundary of src image
64#[derive(Debug, Eq, PartialEq, Copy, Clone)]
65#[repr(usize)]
66pub enum ImageEdgeMode {
67    /// Out of bound pixels are (0,0,0,1) for image with pixel format without alpha channel
68    /// and (0,0,0,0) for image with pixel format that has an alpha channel
69    #[doc(alias = "MPSImageEdgeModeZero")]
70    Zero = 0,
71
72    /// Out of bound pixels are clamped to nearest edge pixel
73    #[doc(alias = "MPSImageEdgeModeClamp")]
74    Clamp = 1,
75
76    /// Out of bound pixels are mirrored wrt. the nearest edge pixel center - ie. the edge of the image is not repeated.
77    /// NOTE: The only filter that currently supports this mode is @ref MPSNNPad - using this with other filters results in undefined behavior.
78    #[doc(alias = "MPSImageEdgeModeMirror")]
79    Mirror,
80
81    /// Out of bound pixels are mirrored wrt. the nearest edge pixel nearest border - ie. the edge of the image is repeated.
82    /// NOTE: The only filter that currently supports this mode is @ref MPSNNPad - using this with other filters results in undefined behavior.
83    #[doc(alias = "MPSImageEdgeModeMirrorWithEdge")]
84    MirrorWithEdge,
85
86    /// Out of bound pixels are filled with a constant value defined by the filter.
87    /// NOTE: The only filter that currently supports this mode is @ref MPSNNPad - using this with other filters results in undefined behavior.
88    #[doc(alias = "MPSImageEdgeModeConstant")]
89    Constant,
90}

92/// Encodes the representation of a single channel within a 'mps::Image'.
93///
94/// A 'mps::Image' pixel may have many channels in it, sometimes many more than 4, the
95/// limit of what 'mtl::PixelFormat's encode. The storage format for a single channel
96/// within a pixel can be given by the MPSImageFeatureChannelFormat. The number
97/// of channels is given by the featureChannels parameter of appropriate 'mps::Image'
98/// APIs. The size of the pixel is size of the channel format multiplied by the
99/// number of feature channels. No padding is allowed, except to round out to a full
100/// byte.
101#[derive(Debug, Eq, PartialEq, Copy, Clone)]
102#[repr(usize)]
103pub enum ImageFeatureChannelFormat {
104    /// No format. This can mean  according to context invalid format or any format.  In the
105    /// latter case, it is an invitation to MPS to pick a format.
106    #[doc(alias = "MPSImageFeatureChannelFormatNone")]
107    None = 0,
108
109    /// u8 with value `[0,255]` encoding `[0,1.0]`
110    #[doc(alias = "MPSImageFeatureChannelFormatUnorm8")]
111    UNorm8 = 1,
112
113    /// u16 with value `[0,65535]` encoding `[0,1.0]`
114    #[doc(alias = "MPSImageFeatureChannelFormatUnorm16")]
115    UNorm16 = 2,
116
117    /// IEEE-754 16-bit floating-point value. "half precision" Representable normal range is +-[2**-14, 65504], 0, Infinity, NaN. 11 bits of precision + exponent
118    #[doc(alias = "MPSImageFeatureChannelFormatFloat16")]
119    F16 = 3,
120
121    /// IEEE-754 32-bit floating-point value.  "single precision" (standard float type in C) 24 bits of precision + exponent
122    #[doc(alias = "MPSImageFeatureChannelFormatFloat32")]
123    F32 = 4,
124
125    /// Reserved for later expansion
126    #[doc(alias = "MPSImageFeatureChannelFormat_reserved0")]
127    Reserved0 = 5,
128
129    /// Always last
130    #[doc(alias = "MPSImageFeatureChannelFormatCount")]
131    Count,
132}

134/// A value to specify a type of data.
135#[doc(alias = "MPSDataType")]
136#[derive(Debug, Eq, PartialEq, Copy, Clone)]
137#[repr(u32)]
138pub enum DType {
139    Invalid,
140
141    /// 32-bit floating point (single-precision).
142    F32 = Self::FLOAT_BIT | 32u32,
143
144    /// 16-bit floating point (half-precision).  (IEEE-754-2008 float16 exchange format)
145    F16 = Self::FLOAT_BIT | 16u32,
146
147    ///Complex number composed of two 32-bit floating point numbers (single-precision).
148    ComplexF32 = Self::FLOAT_BIT | Self::COMPLEX_BIT | 64u32,
149
150    /// Complex number composed of two 16-bit floating point numbers (half-precision).  (IEEE-754-2008 float16 exchange format)
151    ComplexF16 = Self::FLOAT_BIT | Self::COMPLEX_BIT | 32u32,
152
153    /// Signed 8-bit integer.
154    I8 = Self::SIGNED_BIT | 8u32,
155
156    ///  Signed 16-bit integer.
157    I16 = Self::SIGNED_BIT | 16u32,
158
159    ///
160    I32 = Self::SIGNED_BIT | 32u32,
161    I64 = Self::SIGNED_BIT | 64u32,
162
163    /// Unsigned 8-bit integer. Not normalized
164    U8 = 8,
165
166    /// Unsigned 16-bit integer. Not normalized
167    U16 = 16,
168
169    /// Unsigned 32-bit integer. Not normalized
170    U32 = 32,
171    U64 = 64,
172    Bool = Self::ALTERNATE_ENCODING_BIT | 8u32,
173
174    /// Unsigned 1-bit normalized value.
175    UNorm1 = Self::NORMALIZED_BIT | 1u32,
176
177    /// Unsigned 8-bit normalized value.
178    UNorm8 = Self::NORMALIZED_BIT | 8u32,
179}

181impl DType {
182    /// A common bit for all floating point data types. Zero for integer types
183    pub const FLOAT_BIT: u32 = 0x10000000;
184
185    /// A common bit for all complex point data types. Zero for integer types
186    pub const COMPLEX_BIT: u32 = 0x01000000;
187
188    pub const SIGNED_BIT: u32 = 0x20000000;
189
190    pub const ALTERNATE_ENCODING_BIT: u32 = 0x80000000;
191
192    /// If set, the value of the shall be interpreted as value / UNORM_TYPE_MAX
193    /// Normalized values have range [0, 1.0] if unsigned and [-1,1] if signed.
194    /// SNORM_TYPE_MIN is interpreted as SNORM_TYPE_MIN+1 per standard Metal rules.
195    pub const NORMALIZED_BIT: u32 = 0x40000000;
196
197    /// A utility function to get the size of an MPSDataType.  This implementation
198    /// relies upon the specific bit pattern used to encode the type.
199    #[inline]
200    pub fn size_of(&self) -> usize {
201        ((*self as u32 & 0xffffu32) >> 3) as usize
202    }
203}

207/// A description of whether slices,transposes and other views should alias their underlying content or be copied into new storage.
208impl AliasingStrategy {
209    /// View should alias if possible, otherwise make a copy. In the latter case if the allocation fails, nil will be returned.
210    pub const DEFAULT: Self = Self(0);
211
212    pub const DONT_CARE: Self = Self::DEFAULT;
213
214    /// View must alias. If it can not, nil is returned.
215    pub const SHALL_ALIAS: Self = Self(1usize << 0);
216
217    /// Always make a copy. If there isn't enough memory available the allocation may fail and nil will be returned.
218    pub const SHALL_NOT_ALIAS: Self = Self(1usize << 1);
219
220    pub const ALIASING_RESERVED: Self = Self(Self::SHALL_ALIAS.0 | Self::SHALL_NOT_ALIAS.0);
221
222    /// If the view doesn't alias the old object, use temporary for it. If they do alias, the same type of memory must be used.
223    pub const PREFER_TEMPORARY_MEMORY: Self = Self(1usize << 2);
224
225    /// If the view doesn't alias the old object, use non-temporary for it. If they do alias, the same type of memory must be used.
226    pub const PREFER_NON_TEMPORARY_MEMORY: Self = Self(1usize << 3);
227}
