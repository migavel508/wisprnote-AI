21    /// The Metal Command Queue object against which to submit work.
22    ///
23    /// If the client provides a queue, then we will only commit our
24    /// command buffer and let the client handle it's own synchronization
25    /// and/or resource synchronization blits.
26    ///
27    /// If none is provided, then we will use an internal queue which
28    /// automatically commits and waitUntilScheduled.
29    #[doc(alias = "kCARendererMetalCommandQueue")]
30    #[inline]

32        unsafe { kCARendererMetalCommandQueue }
