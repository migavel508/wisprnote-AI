221/// A structure that defines the context or operating environment for an allocator
222/// (cf::Allocator) object. Every Core Foundation allocator object must have a context
223/// defined for it.
224#[repr(C)]
225pub struct AllocatorContext<T = c_void> {
226    pub version: Index,
227
228    /// An untyped pointer to program-defined data. Allocate memory for this data and assign a pointer to it.
229    /// This data is often control information for the allocator. You may assign NULL.
230    pub info: *const T,
231
232    /// A prototype for a function callback that retains the data pointed to by the info field.
233    /// In implementing this function, retain the data you have defined for the allocator context
234    /// in this field. (This might make sense only if the data is a Core Foundation object.)
235    /// You may set this function pointer to None.
236    pub retain: Option<AllocatorRetainCb<T>>,
237
238    /// A prototype for a function callback that releases the data pointed to by the info field.
239    /// In implementing this function, release (or free) the data you have defined for the allocator
240    /// context. You may set this function pointer to None, but doing so might result in memory leaks.
241    pub release: Option<AllocatorReleaseCb<T>>,
242
243    /// A prototype for a function callback that provides a description of the data pointed to
244    /// by the info field. In implementing this function, return a reference to a cf::String object
245    /// that describes your allocator, particularly some characteristics of your program-defined data.
246    /// You may set this function pointer to None, in which case Core Foundation will provide a rudimentary
247    /// description.
248    pub copy_description: Option<AllocatorCopyDescCb<T>>,
249
250    /// A prototype for a function callback that allocates memory of a requested size.
251    /// In implementing this function, allocate a block of memory of at least size
252    /// bytes and return a pointer to the start of the block. The hint argument is
253    /// a bitfield that you should currently not use (that is, assign 0). The size parameter
254    /// should always be greater than 0. If it is not, or if problems in allocation occur,
255    /// return NULL. This function pointer may not be assigned NULL.
256    pub allocate: AllocatorAllocateCb<T>,
257
258    /// A prototype for a function callback that reallocates memory of a requested size
259    /// for an existing block of memory.
260    pub reallocate: AllocatorRealloacteCb<T>,
261
262    /// A prototype for a function callback that deallocates a given block of memory.
263    /// In implementing this function, make the block of memory pointed to by ptr available
264    /// for subsequent reuse by the allocator but unavailable for continued use by the program.
265    /// The ptr parameter cannot be NULL and if the ptr parameter is not a block of memory that
266    /// has been previously allocated by the allocator, the results are undefined; abnormal
267    /// program termination can occur. You can set this callback to None, in which case the
268    /// CFAllocatorDeallocate function has no effect.
269    pub deallocate: Option<AllocatorDealloacteCb<T>>,
270
271    /// A prototype for a function callback that determines whether there is enough free memory
272    /// to satisfy a request. In implementing this function, return the actual size the allocator
273    /// is likely to allocate given a request for a block of memory of size size. The hint argument
274    /// is a bitfield that you should currently not use.
275    pub preferred_size: AllocatorPreferredSizeCb<T>,
276}

440#[cfg(test)]
441mod tests {
442    use std::ffi::c_void;
443
444    use crate::{arc, cf};
445
446    #[test]
447    fn custom_allocator_with_padding() {
448        let system = cf::Allocator::system_default().unwrap();
449        const PADDING: isize = 10isize;
450
451        extern "C" fn retain(info: *const cf::Allocator) -> *const cf::Allocator {
452            eprintln!("retain");
453            unsafe {
454                let rf = info.as_ref().unwrap().retained();
455                let res = rf.as_ref() as *const cf::Allocator;
456                std::mem::forget(rf);
457                res
458            }
459        }
460        extern "C" fn release(info: *const cf::Allocator) {
461            eprintln!("release");
462            let retained: arc::R<cf::Allocator> = unsafe { std::mem::transmute(info) };
463            std::mem::drop(retained);
464        }
465
466        extern "C" fn copy_desc(info: *const cf::Allocator) -> Option<arc::R<cf::String>> {
467            unsafe { info.as_ref().map(|a| a.desc()) }
468        }
469
470        extern "C" fn allocate(
471            size: cf::Index,
472            hint: cf::OptionFlags,
473            info: *mut cf::Allocator,
474        ) -> *mut c_void {
475            unsafe {
476                info.as_ref()
477                    .map(|a| a.allocate(size + PADDING, hint).offset(PADDING))
478                    .unwrap_or(std::ptr::null_mut())
479            }
480        }
481
482        extern "C" fn reallocate(
483            ptr: *mut c_void,
484            new_size: cf::Index,
485            hint: cf::OptionFlags,
486            info: *mut cf::Allocator,
487        ) -> *mut c_void {
488            unsafe {
489                info.as_ref()
490                    .map(|a| {
491                        a.reallocate(ptr.offset(-PADDING), new_size + PADDING, hint)
492                            .offset(PADDING)
493                    })
494                    .unwrap_or(std::ptr::null_mut())
495            }
496        }
497
498        extern "C" fn deallocate(ptr: *mut c_void, info: *mut cf::Allocator) {
499            unsafe { info.as_ref().unwrap().deallocate(ptr.offset(-PADDING)) }
500        }
501
502        extern "C" fn preferred_size(
503            size: cf::Index,
504            hint: cf::OptionFlags,
505            info: *mut cf::Allocator,
506        ) -> cf::Index {
507            unsafe { info.as_ref().unwrap().preferred_size(size + PADDING, hint) - PADDING }
508        }
509
510        let mut context = cf::AllocatorContext::<cf::Allocator> {
511            version: 0,
512            info: system as *const _,
513            retain: Some(retain),
514            release: Some(release),
515            copy_description: Some(copy_desc),
516            allocate,
517            reallocate,
518            deallocate: Some(deallocate),
519            preferred_size,
520        };
521
522        let alloc = cf::Allocator::new(&mut context).unwrap();
523
524        assert_eq!(alloc.preferred_size(10, cf::OptionFlags::NONE), 10);
525
526        let mem = unsafe { alloc.allocate(10, cf::OptionFlags::NONE) };
527        assert!(!mem.is_null());
528        let buf: *mut u8 = mem.cast();
529        let slice = unsafe { std::slice::from_raw_parts_mut(buf, 10) };
530        slice[0] = 10;
531        let bigger =
532            unsafe { std::slice::from_raw_parts_mut(buf.offset(-PADDING), 10 + PADDING as usize) };
533
534        bigger[0] = 1;
535        assert_eq!(bigger[PADDING as usize], 10);
536        unsafe { alloc.deallocate(mem) };
537    }
