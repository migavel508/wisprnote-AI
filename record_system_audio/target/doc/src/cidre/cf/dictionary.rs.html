263    pub fn keys(&self) -> Vec<&Type> {
264        let len = self.len();
265        let mut keys: Vec<&Type> = Vec::with_capacity(len);
266        unsafe {
267            let keys_ptr = keys.as_ptr() as *mut *const c_void;
268            self.keys_and_values(keys_ptr, std::ptr::null_mut());
269            keys.set_len(len);
270        }
271        keys
272    }

290        let len = self.len();
291        let mut values: Vec<&Type> = Vec::with_capacity(len);
292        unsafe {
293            let values_ptr = values.as_ptr() as *mut *const c_void;

295            values.set_len(len);
296        }

300    pub fn keys_with_values(&self) -> (Vec<&Type>, Vec<&Type>) {
301        let len = self.len();
302        let mut keys: Vec<&Type> = Vec::with_capacity(len);
303        let mut values: Vec<&Type> = Vec::with_capacity(len);
304        unsafe {
305            let keys_ptr = keys.as_ptr() as *mut *const c_void;
306            let values_ptr = values.as_ptr() as *mut *const c_void;
307            self.keys_and_values(keys_ptr, values_ptr);
308            keys.set_len(len);
309            values.set_len(len);
310        }
311        (keys, values)
312    }
