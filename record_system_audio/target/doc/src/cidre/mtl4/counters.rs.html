18#[doc(alias = "MTL4TimestampGranularity")]
19#[derive(Debug, Eq, PartialEq, Copy, Clone)]
20#[repr(isize)]
21pub enum TimestampGranularity {
22    /// A minimally-invasive timestamp which may be less precise.
23    ///
24    /// Using this granularity incurs in the lowest overhead, at the cost of precision. For example, it may sample at
25    /// command encoder boundaries.
26    Relaxed = 0,
27
28    /// A timestamp as precise as possible.
29    ///
30    /// Using this granularity may incur in a performance penalty, for example, it may cause splitting of command encoders.
31    Precise = 1,
32}
